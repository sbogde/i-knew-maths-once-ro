\noindent\fbox{T} \ Orice grup $G$ este izomorf cu un grup de permutări.

\bigskip
\noindent\underline{Dem}: \ Fie $a\in G$.

\noindent $\theta_a: G\to G$. \quad $\underline{\theta_a(x)=ax}$, \quad $\underline{\theta_a\in S_G}$.

\[
\theta_a(x_1)=\theta_a(x_2) \Rightarrow ax_1=ax_2 \Rightarrow x_1=x_2
\]
\[
(\forall)\,y\in G \ \exists\,x\in G \text{ a.î. } \theta(x)=y. \quad \text{(iau } x=a^{-1}y\text{)}
\]

\bigskip
\noindent\underline{$\theta_a\circ\theta_b = \theta_{ab}$}.

\[
(\theta_a\circ\theta_b)(x) = \theta_a\big(\theta_b(x)\big) = \theta_a(bx) = abx = \theta_{ab}(x).
\]

\bigskip
\noindent Fie $f: G \to (S_G,\circ,1_G)$.

\[
f(a) = \theta_a
\]
\[
f(ab) = \theta_{ab} = \theta_a\circ\theta_b = f(a)\cdot f(b)
\]

\bigskip
\noindent\underline{$f$ e inj}
