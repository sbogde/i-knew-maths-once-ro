\section*{Curs 9 - Pagina 8}

\noindent\underline{Dem.}

\bigskip
\noindent (1) Dacă $a,b\in\operatorname{Stab}_G(x)$, atunci
\[
\left.\begin{aligned}
 ax&=x,\\
 bx&=x
\end{aligned}\right\}
\quad\Longrightarrow\quad (ab)x=a(bx)=ax=x.
\]
\[
a^{-1}x=a^{-1}(ax)=(a^{-1}a)x=ex=x.
\]
\[
\Longrightarrow\quad \operatorname{Stab}_G(x)\leq G.
\]

\bigskip
\noindent (3) $\Longleftarrow$ (2).

\bigskip
\noindent (2) $\operatorname{Orb}(x)=Gx=\{ax\mid a\in G\}$.

\bigskip
\noindent Fie $a,b\in G$. În ce cond. $ax=bx$?

\bigskip
\noindent Dar $ax=bx\ \Longleftrightarrow\ a^{-1}b\in\operatorname{Stab}_G(x)\leq G$.
\[
\big(\Longleftrightarrow\ a^{-1}bx=x\ \Longleftrightarrow\ bx=ax\big).
\]
