\bigskip
\noindent\underline{(2) Teoreme de izomorfisme}

\medskip
\noindent\fbox{T I de izom.}

\bigskip
\noindent Fie $G$ grup, $H\leq G$, $N\trianglelefteq G$.
\[
\text{Atunci}\qquad \boxed{\frac{HN}{N}\cong\frac{H}{H\cap N}}.
\]

\noindent (Atunci $HN=NH$, $HN\leq G$, $N\trianglelefteq HN$,
$H\cap N\trianglelefteq H$ \c{s}i \ldots)
% Ultimul segment al parantezei este prescurtat in manuscris.

\medskip
\noindent $(G,+)\ \Rightarrow\ \dfrac{H+N}{N}\cong\dfrac{H}{H\cap N}$.

\bigskip
\noindent\underline{Dem.}
Fie $xy\in HN$, $x\in H$, $y\in N$.
% In manuscris exista mai multe incercari barate pentru rescrierea lui xy.
\[
xy=(xyx^{-1})x\in NH,
\qquad
\underbrace{xyx^{-1}}_{\in N}\ \underbrace{x}_{\in H}.
\]
\noindent Analog, $NH\subseteq HN$; deci $HN=NH$.

\bigskip
\noindent Fie $x_1y_1,\,x_2y_2\in HN=NH$, cu
$x_1,x_2\in H$, $y_1,y_2\in N$.
\[
\begin{aligned}
x_1y_1x_2y_2
 &=\underbrace{x_1x_2}_{\in H}
   \underbrace{(x_2^{-1}y_1x_2)y_2}_{\in N}
   \in HN.
\end{aligned}
\]

\[
(x_1y_1)^{-1}=y_1^{-1}x_1^{-1}\in NH=HN.
\]

\noindent Evident, $H,N\subseteq HN$.

\[
\left.\begin{gathered}
N\trianglelefteq G,\\
HN\leq G,\\
N\subseteq HN
\end{gathered}\right\}
\ \Longrightarrow\ N\trianglelefteq HN.
\]

\noindent $H\cap N\trianglelefteq H$
(va ap\u{a}rea ca nucleul unui morfism de grup).
