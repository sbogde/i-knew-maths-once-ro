\noindent\fbox{T} \ $(\forall)\,a,b\in\mathbb{N}$, $\exists!$ c.m.m.d.c.\ al lui $a$ şi $b$.

\bigskip
\noindent\underline{Lemă}: Fie $a,b,q,r\in\mathbb{N}$ a.î.\ $a=bq+r$.

\noindent $\exists\,(a,b) \Leftrightarrow \exists\,(b,r)$ şi în acest caz $(a,b)=(b,r)$.

\bigskip
\noindent\underline{Dem}

\noindent $d\mid a$, $d\mid b \Rightarrow d\mid b$, $d\mid r$

\bigskip
\noindent\underline{Ob}: dacă $a\mid b \Rightarrow \exists\,(a,b)=a$; $(a,0)=a$ $(\forall)\,a$.

\bigskip
\noindent\underline{Dem}: Pp.\ $b\nmid a$ \quad $b\neq 0$.

\noindent $a=bq_0+r_0$, \ $0<r_0<b$.

\noindent Dacă $r_0\neq 0 \Rightarrow b=r_0q_1+r_1$, \ $0<r_1<r_0$ \ (Alg.\ lui Euclid)

\[
r_0=r_1q_2+r_2.
\]
\[
\vdots
\]
\[
r_{k-2}=r_{k-1}q_k+r_k.
\]

\bigskip
\noindent\underline{Corolar}:

\noindent $P=\{r_0,r_1,\ldots,r_k,\ldots\}$; $P\neq\varnothing$, $P\subseteq\mathbb{N}$.

\noindent Fie $r_n\in P$, $r_n\leq r_i$, $(\forall)\,i\in\mathbb{N} \Rightarrow r_{n+1}=0$.

\bigskip
\noindent\underline{Consecinţă}:

\[
(ca,cb)=c(a,b)
\]
