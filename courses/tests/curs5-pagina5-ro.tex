\noindent\underline{\textbf{Corolar}}

\bigskip
\noindent $N\trianglelefteq G \Rightarrow R_N$ e congruență.

\bigskip
\noindent\underline{Def}: Dacă $G$=grup şi $N\trianglelefteq G \Rightarrow$ grupul $G/R_N \overset{\text{not}}{=} G/N$ s.n.

\noindent grupul factor al lui $G$ prin subgrupul normal $N$.

\bigskip
\noindent $A\neq\varnothing$, \ $\varphi: A\times A\to A$ lege de comp.\ pe $A$.

\noindent $R$=rel.\ de ech.\ pe $A$ compat.\ cu $\varphi$, \quad $xRx'$ şi $yRy' \Rightarrow \varphi(x,y)R\varphi(x',y')$

\noindent = congruență pe $A$.

\[
\frac{A}{R} = \{\hat a \mid a\in A\}, \qquad \varphi^*: \frac{A}{R}\times\frac{A}{R} \to \frac{A}{R}:
\]
\[
\varphi^*(\hat a,\hat b) = \varphi(a,b) \qquad \hat a\hat b=\widehat{ab} \ \text{(not.\ multiplicativă)}
\]

\bigskip
\noindent $\varphi$ = asoc, comut $\Rightarrow \varphi^*$ = asoc, comut.

\noindent $\varphi$ are el.\ neutru pe $A \Rightarrow \hat e$ el.\ neutru pt.\ $\varphi^*$.

\noindent $a\in A$ are simetric $a'\in A$ în rap.\ cu $\varphi \Rightarrow \hat a\in \dfrac{A}{R}$ are simetric

\noindent $\hat{a'}\in \dfrac{A}{R}$ în rap.\ cu $\varphi^*$.

\bigskip
\noindent Deci, dacă $(A,\varphi)$=grup $\Rightarrow \Big(\dfrac{A}{R},\varphi^*\Big)$ = grup.

\bigskip
\noindent pe $G/N$: \quad $\hat a\hat b = \widehat{ab}$.

\[
aN\cdot bN = abN.
\]

\noindent $\hat e = N$.
