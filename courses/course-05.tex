\noindent\underline{\textbf{T.\ lui Lagrange}}

\bigskip
\noindent Fie $(G,\cdot)$ grup.\ $H\leq G$.

\noindent Dacă $x,y\in G$ \quad $x\equiv_s y \pmod H \overset{\text{def}}{\Longleftrightarrow} x^{-1}y\in H$.

\[
R_H^s = \{(x,y)\in G\times G \mid x^{-1}y\in H\} \subseteq G\times G.
\]

\noindent $R_H^s$ = rel.\ de ech.\ pe $G$, ce e compatibilă la st.\ cu $\cdot$.

\[
xR_H^s y \Rightarrow zxR_H^s zy
\]

\noindent Pt.\ $a\in G$, \ $[a]_{R_H^s} = aH = \{ah \mid h\in H\}$ = clasa de resturi la stânga

\bigskip
\noindent $x\equiv_d y \pmod H \overset{\text{def}}{\Longleftrightarrow} xy^{-1}\in H$.

\[
R_H^d = \{(x,y)\in G\times G \mid xy^{-1}\in H\} = \text{rel.\ de ech.\ compat.\ la dr.\ cu } \cdot.
\]

\[
xR_H^d y \Rightarrow xzR_H^d yz.
\]

\[
[a]_{R_H^d} = Ha = \{ha \mid h\in H\}.
\]

\bigskip
\noindent Deci $T=\{a_i\}_{i\in I}$ = transversală la stânga a lui $H$ în $G$

\noindent = sist.\ complet de reprezentanți

\begin{enumerate}
\item[1)] $i\neq j \Rightarrow a_i \not\equiv_s a_j \pmod H$
\item[2)] $(\forall)\, x\in G$ \ $\exists\, i\in I$ a.î.\ $x\equiv_s a_i \pmod H$
\end{enumerate}

\bigskip
\noindent\underline{Not.} $T^{-1} = \{a_i^{-1}\}_{i\in I}$

\noindent $T$ = transv.\ la stânga $\Rightarrow$ $T^{-1}$ = transv.\ la dreapta.

\[
T \xrightarrow{\ } T^{-1}, \qquad a_i \mapsto a_i^{-1} \text{ e bijecție} \ \Rightarrow \ \text{card}\, T = \text{card}\, T^{-1}
\]

\noindent = indicele lui $H$ în $G \overset{\text{not}}{=} [G:H]$

\noindent ord $G$ = card $G$ = $|G|$.

\bigskip
\noindent\fbox{T.\ Lagrange} \quad Dacă $(G,\cdot)$ = grup finit şi $H$ un subgr.\ al lui $G$,

\noindent atunci:

\[
\boxed{|G| = |H|\cdot [G:H]}
\]

\noindent sau, atunci: \quad $\big($ord $G$ = ord $H\cdot [G:H]\big)$

\bigskip
\noindent\underline{Aplicație:} ord $S_n = n!$

\[
S_n = \{\sigma : \{1,2,\ldots,n\}\to\{1,2,\ldots,n\} \mid \sigma \text{ = bijectivă}\}.
\]

% \noindent [Notăm:]\footnote{fragmentul de legătură de aici e foarte compactat în original.}
\[
\sigma = \begin{pmatrix} 1 & 2 & \cdots & n \\ \sigma(1) & \sigma(2) & \cdots & \sigma(n) \end{pmatrix}
\]

\noindent Inducție după $n$.

\bigskip
\noindent $n=1$. \quad ord $S_1=1$.

\bigskip
\noindent Pp.\ ord $S_{n-1}=(n-1)!$.

\noindent\underline{ord $S_n=n!$}

\bigskip
\noindent Fie $\sigma_i\in S_n$ a.î.\ $\sigma_i(n)=i$, $(\forall)\, i=\overline{1,n}$.

\bigskip
\noindent $H=\{\sigma\in S_n \mid \sigma(n)=n\}$. \quad 1) \underline{$H\leq S_n$}, [ord $\underline{H=(n-1)!}$]

\bigskip
\noindent (3) $\{\sigma_1,\sigma_2,\ldots,\sigma_n\}$ transversală la stânga (şi la dreapta) a lui $H$ în $S_n$.

\bigskip
\noindent (1) $\sigma,\bar\alpha\in H$, $H\neq\varnothing$,

\[
\sigma(\bar\alpha(n))=\sigma(n)=n \ \Rightarrow \ \sigma\circ\bar\alpha\in H \ \Rightarrow \ H = \text{p.s.}
\]

\noindent $\sigma\in H \Rightarrow \sigma(n)=n \ \Rightarrow \ \sigma^{-1}(\sigma(n))=\sigma^{-1}(n) \ \Rightarrow \ n=\sigma^{-1}(n) \ \Rightarrow$

\[
\Rightarrow \ \sigma^{-1}\in H \ \Rightarrow \ \underline{H\leq S_n}.
\]

\bigskip
\noindent (2) \quad $S_{n-1} \xrightarrow{\ f\ } S_n$

\[
\alpha = \begin{pmatrix} 1 & 2 & \cdots & n-1 \\ \alpha(1) & \alpha(2) & \cdots & \alpha(n-1) \end{pmatrix}
\]
\[
f(\alpha) = \begin{pmatrix} 1 & 2 & \cdots & n-1 & n \\ \alpha(1) & \alpha(2) & \cdots & \alpha(n-1) & n \end{pmatrix} \in H.
\]

\noindent $f$ bij.\ între $S_{n-1}$ şi $H$ -- evident.

\[
\Rightarrow S_{n-1}\cong H \overset{\text{ip.\ de inducție}}{\Longrightarrow} \text{ord } H=(n-1)!.
\]

\bigskip
\noindent (3) \quad $\big[i\neq j \Rightarrow \sigma_i \not\equiv_s \sigma_j \pmod H\big]$.

\noindent $\big[(\forall)\, \sigma\in S_n \ \exists!\, i \text{ a.î. } \sigma\equiv_s \sigma_i \pmod H\big]$.

\bigskip
\noindent Pp.\ că $\sigma_i\equiv_s\sigma_j \pmod H \Rightarrow \sigma_i^{-1}\circ\sigma_j\in H$

\noindent $\Rightarrow \sigma_i^{-1}(\sigma_j(n))=n. \Rightarrow \sigma_j(n)=\sigma_i(n) \Rightarrow j=i$

\bigskip
\noindent Fie $i=\sigma(n)$. Afirm că $\sigma\equiv_s\sigma_i \pmod H \Leftrightarrow \sigma^{-1}\sigma_i\in H$.

\[
\sigma^{-1}(\sigma_i(n))=n \Leftrightarrow \underbrace{\sigma_i(n)}_{i} = \underbrace{\sigma(n)}_{i} \quad \text{adev.}
\]

\bigskip
\noindent\underline{Lagrange} $\Rightarrow$ ord $S_n$ = ord $H\cdot[S_n:H]$ = ord $H\cdot$ord$\{\sigma_1,\ldots,\sigma_n\}$

\[
= (n-1)!\cdot n.
\]
